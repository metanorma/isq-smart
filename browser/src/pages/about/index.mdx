export const qtyParts = getPartsByDomain('quantities')
export const mathParts = getPartsByDomain('math')

export const knownTitle = (base) => base === '2' ? 'Mathematical Notation' : base === '11' ? 'Characteristic Numbers' : undefined

export function groupScopeParts(parts) {
  const groups = new Map()
  for (const p of parts) {
    const base = basePartKey(p.partKey)
    const meta = getPartMeta(base)
    const g = groups.get(base)
    if (g) {
      g.entries = g.entries + partEntryCount(p.partKey)
    } else {
      const title = (meta && meta.title) || knownTitle(base) || p.title
      const icon = (meta && meta.icon) || p.icon
      groups.set(base, { partKey: base, title: title, icon: icon, entries: partEntryCount(p.partKey) })
    }
  }
  return Array.from(groups.values()).sort((a, b) => parseInt(a.partKey, 10) - parseInt(b.partKey, 10))
}

export const scopeParts = groupScopeParts([...qtyParts, ...mathParts])
